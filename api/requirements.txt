# Runtime dependencies for the Lambda container (api/Dockerfile), Python 3.11.
#
# The AWS python:3.11 base image is Amazon Linux 2 (glibc 2.26). These are the newest
# versions of the scientific stack that still publish glibc-2.17 (manylinux2014)
# wheels, so pip installs prebuilt binaries instead of trying -- and failing, with no
# compiler in the image -- to build from source. They are ALSO the exact versions the
# model was trained with (under Python 3.11), so the pickle loads with no version
# mismatch. numpy/scipy/joblib are pinned alongside scikit-learn because they all
# participate in that pickle.
scikit-learn==1.7.2
numpy==2.2.6
scipy==1.16.3
joblib==1.5.3
pandas==2.3.2

# Web layer + Lambda adapter. Pin starlette/pydantic (fastapi's transitive deps) too:
# leaving them to float let the Linux image resolve a NEWER, incompatible Starlette than
# the one tested locally, which crashed the container at import (app.add_middleware).
# These are the exact versions verified working end-to-end.
fastapi==0.141.1
starlette==1.4.1
pydantic==2.13.4
pydantic-core==2.46.4
anyio==4.14.2
mangum==0.21.0
requests==2.34.2

# Chat assistant (api/chat.py). Pure Python except jiter, its JSON parser, which is a
# Rust extension -- verified to publish a manylinux_2_17 (glibc 2.17) wheel, so it
# installs prebuilt on this Amazon Linux 2 base like everything above. anthropic needs
# pydantic>=1.9,<3 and anyio, both already pinned above at compatible versions.
anthropic==0.122.0

# PostHog AI observability for the server-side Anthropic call (api/chat.py). Pure Python,
# no compiled extension, so it installs prebuilt on the Amazon Linux 2 base like the rest.
# posthog.ai.anthropic wraps the client to report model, latency, tokens, cost, errors and
# tool calls; the question and answer are held back by privacy mode. Optional at runtime:
# with POSTHOG_PROJECT_API_KEY unset the assistant answers as before and records nothing.
posthog==7.45.1
